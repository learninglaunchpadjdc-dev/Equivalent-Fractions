    <script>

        const canvas = document.getElementById('visualCanvas');

        const ctx = canvas.getContext('2d');

        const container = document.getElementById('canvasContainer');

        const inputs = ['numA', 'denA', 'numB', 'denB'];

        const resultBox = document.getElementById('resultBox');

        const resultText = document.getElementById('resultText');

        const ratioDisplay = document.getElementById('ratioDisplay');

        const logsContainer = document.getElementById('logs');


        function init() {

            window.addEventListener('resize', resize);

            inputs.forEach(id => {

                document.getElementById(id).addEventListener('input', update);

            });

            resize();

            update();

        }


        function resize() {

            const rect = container.getBoundingClientRect();

            const dpr = window.devicePixelRatio || 1;

            canvas.style.width = rect.width + 'px';

            canvas.style.height = rect.height + 'px';

            canvas.width = rect.width * dpr;

            canvas.height = rect.height * dpr;

            ctx.scale(dpr, dpr);

            update();

        }


        function addLog(msg, type = 'info') {

            const log = document.createElement('div');

            log.className = `p-2 rounded bg-black/20 border-l-2 ${type === 'success' ? 'border-emerald-500 text-emerald-300' : 'border-slate-500 text-slate-400'}`;

            log.innerHTML = `<span class="opacity-50">[${new Date().toLocaleTimeString([], {hour12:false})}]</span> ${msg}`;

            logsContainer.prepend(log);

            if(logsContainer.children.length > 5) logsContainer.lastChild.remove();

        }


        function update() {

            const na = parseInt(document.getElementById('numA').value) || 0;

            const da = parseInt(document.getElementById('denA').value) || 1;

            const nb = parseInt(document.getElementById('numB').value) || 0;

            const db = parseInt(document.getElementById('denB').value) || 1;


            const valA = na / da;

            const valB = nb / db;

            const isEquivalent = Math.abs(valA - valB) < 0.000001 && da !== 0 && db !== 0;


            // UI State

            if (isEquivalent) {

                resultBox.className = "mt-10 w-full py-4 rounded-lg text-center border border-emerald-500/50 bg-emerald-500/10 text-emerald-400";

                resultText.innerText = "EQUIVALENT";

                ratioDisplay.innerText = `RATIO: ${valA.toFixed(3)}`;

            } else {

                resultBox.className = "mt-10 w-full py-4 rounded-lg text-center border border-rose-500/50 bg-rose-500/10 text-rose-400";

                resultText.innerText = "NOT EQUIVALENT";

                ratioDisplay.innerText = `DIFF: ${Math.abs(valA - valB).toFixed(3)}`;

            }


            draw(na, da, nb, db);

        }


        function draw(na, da, nb, db) {

            const dpr = window.devicePixelRatio || 1;

            const w = canvas.width / dpr;

            const h = canvas.height / dpr;

            ctx.clearRect(0, 0, w, h);


            const barHeight = 60;

            const barWidth = w * 0.8;

            const startX = (w - barWidth) / 2;

            const startY = h / 2 - 80;


            // Draw Bar A

            drawFractionBar(startX, startY, barWidth, barHeight, na, da, "FRACTION A", "#EAA834");

            

            // Draw Bar B

            drawFractionBar(startX, startY + 120, barWidth, barHeight, nb, db, "FRACTION B", "#4ade80");

        }


        function drawFractionBar(x, y, w, h, num, den, label, color) {

            ctx.fillStyle = "rgba(255,255,255,0.05)";

            ctx.fillRect(x, y, w, h);


            // Label

            ctx.fillStyle = "rgba(255,255,255,0.4)";

            ctx.font = '10px "JetBrains Mono"';

            ctx.textAlign = 'left';

            ctx.fillText(label, x, y - 10);


            if (den <= 0) return;


            const segmentWidth = w / den;

            

            // Draw Segments

            for (let i = 0; i < den; i++) {

                ctx.strokeStyle = "rgba(255,255,255,0.1)";

                ctx.strokeRect(x + (i * segmentWidth), y, segmentWidth, h);

                

                if (i < num) {

                    ctx.fillStyle = color;

                    ctx.globalAlpha = 0.6;

                    ctx.fillRect(x + (i * segmentWidth) + 2, y + 2, segmentWidth - 4, h - 4);

                    ctx.globalAlpha = 1.0;

                }

            }

        }


        init();
